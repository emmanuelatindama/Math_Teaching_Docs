\begin{enumerate}[1.]
  \item For a 2×2 system:
    \[
    \begin{cases}
    x_1' = 3 x_1 + 4 x_2 + e^{2t},\\
    x_2' = -x_1 + x_2 + \sin t,
    \end{cases}
    \]
    \begin{itemize}
      \item[(a)] Solve homogeneous system:
      \(\mathbf{x}_h' = A \mathbf{x}_h, \quad A = \begin{bmatrix}3 & 4 \\ -1 & 1\end{bmatrix}\).
      \item[(b)] Find particular solution $\mathbf{x}_p(t)$ using undetermined coefficients or variation of parameters.
      \item[(c)] General solution: \(\mathbf{x}(t) = \mathbf{x}_h(t) + \mathbf{x}_p(t)\).
    \end{itemize}
  \vspace{1cm}

  \item Solve $\mathbf{x}' = \begin{bmatrix} 2 & 1 \\ 0 & 3 \end{bmatrix} \mathbf{x}$, $\mathbf{x}(0) = \begin{bmatrix} 1 \\ 0 \end{bmatrix}$.
  \vspace{1cm}

  \item Convert the IVP
        \[
        y'' - 3y' + 2y = 0, \quad y(0)=1, \ y'(0)=0
        \]
        into a first-order system.

\end{enumerate}
