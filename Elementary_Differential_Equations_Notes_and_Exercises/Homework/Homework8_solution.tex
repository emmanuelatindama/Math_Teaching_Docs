\begin{enumerate}
  \item[\textbf{Q1}] Solve the following using the superposition method:
    \begin{enumerate}[(a)]
      \item $y''+y=5\cos(2x)$
         \begin{center}\setlength{\fboxsep}{10pt}\fcolorbox{yellow!20}{yellow!20}{\parbox{0.9\linewidth}{
        \textbf{Solution}
            Homogeneous solution:
          \[
          y_h=C_1\cos x + C_2\sin x.
          \]
          Particular solution, try
          \[
          y_p=A\cos(2x)+B\sin(2x).
          \]
          Substitute:
          \[
          y_p''+y_p=(-3A)\cos(2x)+(-3B)\sin(2x)=5\cos(2x)
        \]
          so \(A=-\tfrac{5}{3},; B=0\). Hence
          \[
          y=C_1\cos x + C_2\sin x -\tfrac{5}{3}\cos(2x)
        \]
        }}
        \end{center}

      \item $y''-y=xe^x$
         \begin{center}\setlength{\fboxsep}{10pt}\fcolorbox{yellow!20}{yellow!20}{\parbox{0.9\linewidth}{
        \textbf{Solution}

        \[
        y_h=C_1 e^{x}+C_2 e^{-x}.
       \]
        RHS is (\(\text{linear polynomial})\cdot e^{x}\).
        Since \(e^{x}\) is a solution of the homogeneous equation (root \(r=1\) of multiplicity \(1\)), multiply the usual trial by \(x\). Trial:
        \[
        y_p = x(Ax+B)e^{x}=(A x^2+ B x)e^{x}.
       \]
        Compute (as in text) to find \(A=\tfrac14,;B=-\tfrac14\). Thus
        \[
        \boxed{,y=C_1 e^{x}+C_2 e^{-x}+\frac{x^2-x}{4}e^{x},.}
       \]
        }}
        \end{center}
      \item $y''+4y'+4y=e^{-2x}+x$
         \begin{center}\setlength{\fboxsep}{10pt}\fcolorbox{yellow!20}{yellow!20}{\parbox{0.9\linewidth}{
        \textbf{Solution}:
        
        Homogeneous solution (double root (r=-2)):
          \[
          y_h=(C_1 + C_2 x)e^{-2x}.
         \]
          Treat RHS by superposition.

          For \(e^{-2x}\): resonance with root \(r=-2\) of multiplicity \(2\) so multiply by \(x^2\). Trial
          \[
          y_{p1}=A x^2 e^{-2x}.
         \]
          Applying \((D+2)^2\) gives \(2A e^{-2x}\), so \(A=\tfrac12\); hence \(y_{p1}=\tfrac12 x^2 e^{-2x}\).

          For polynomial \(x\), try
          \[
          y_{p2}=ax+b.
         \]
          Substituting yields \(a=\tfrac14,; b=-\tfrac14\)

          Combine
          \[
          \boxed{y=(C_1 + C_2 x)e^{-2x} + \tfrac12 x^2 e^{-2x} + \tfrac14 x -\tfrac14.}
         \]
        }}
        \end{center}
      \end{enumerate}

  \vspace{0.5cm}

  \item[\textbf{Q2}] For each of the following, determine the homogeneous solution and the \textbf{form} (\textcolor{red}{do not solve}) of a 
    particular non-homogeneous solution:
      \begin{enumerate}[(i)]
        \item \(y'' + 4y = 2x^{2}\)
        \item \(y'' + 4y = 10 + \sin(x)\)
        \item \(y'' + 4y = \sin(2 x)\)
        \item \(y'' + 4y = e^{x}\sin(2 x)\)
      \end{enumerate}

    \begin{center}\setlength{\fboxsep}{10pt}\fcolorbox{yellow!20}{yellow!20}{\parbox{0.9\linewidth}{
    \textbf{Solution}
    
      (i) Particular \emph{form}: polynomial of degree 2,
      \[
      y_p = Ax^2+Bx+C.
      \]
      (ii) \(y_p\) is sum of a constant and a sinusoid:
      \[
      y_p = A + \left(B\cos x + C\sin x\right).
      \]
      (iii) Particular:  sinusoid,
      \[
      y_p = A\cos(2x)+B\sin(2x).
      \]
      (iv) Particular: exponential times sinusoid,
      \[
      y_p = e^{x}\left(A\cos(2x)+B\sin(2x)\right).
      \]
      }}
    \end{center}

  \item[\textbf{Q3}] Find the general solution of \(y'' + 4y = x\cos{x}\) using the variation of parameters approach.
  \begin{center}\setlength{\fboxsep}{10pt}\fcolorbox{yellow!20}{yellow!20}{\parbox{0.9\linewidth}{
    \textbf{Solution}
    
      Find a inhomogeneous solution of
      \[
      y'' + 4y = x\cos x
      \]
      by variation of parameters.

      The homogeneous equation \(y''+4y=0\) has fundamental solutions
      \[
      y_1=\cos 2x,\qquad y_2=\sin 2x,
      \]
      with Wronskian
      \[
      W=\begin{vmatrix}\cos2x & \sin2x\\ -2\sin2x & 2\cos2x\end{vmatrix}
      =2(\cos^2 2x+\sin^2 2x)=2.
      \]

      Variation of parameters gives
      \[
      u_1'(x) = -\frac{f(x)\,y_2(x)}{W} = -\frac{x\cos x\;\sin2x}{2},\qquad
      u_2'(x) = \frac{f(x)\,y_1(x)}{W} = \frac{x\cos x\;\cos2x}{2},
      \]
      so
      \[
      u_1(x) = -\tfrac{1}{2}\int x\cos x\sin2x\,dx,\qquad
      u_2(x)=\tfrac{1}{2}\int x\cos x\cos2x\,dx.
      \]

      Use product-to-sum identities:
      \[
      \cos x\cos2x=\tfrac12(\cos3x+\cos x),\qquad
      \cos x\sin2x=\tfrac12(\sin3x+\sin x).
      \]
      Thus
      \[
      u_2'(x)=\frac{x}{4}(\cos3x+\cos x),\qquad
      u_1'(x)=-\frac{x}{4}(\sin3x+\sin x).
      \]
      Integrate termwise (integration by parts for \(\int x\cos kx\,dx\) and \(\int x\sin kx\,dx\)):

      \[
      \int x\cos kx\,dx=\frac{x\sin kx}{k}+\frac{\cos kx}{k^2},\qquad
      \int x\sin kx\,dx=-\frac{x\cos kx}{k}+\frac{\sin kx}{k^2}.
      \]

      Hence (dropping arbitrary constants)
      \[
      \begin{aligned}
      u_2(x) &= \tfrac{1}{4}\Big[\tfrac{1}{3}x\sin 3x + \tfrac{1}{9}\cos 3x \;+\; x\sin x + \cos x\Big],\\[4pt]
      u_1(x) &= \tfrac{1}{4}\Big[\tfrac{1}{3}x\cos 3x - \tfrac{1}{9}\sin 3x \;+\; x\cos x - \sin x\Big].
      \end{aligned}
      \]

      Therefore a particular solution is
      \[
      y_p(x)=u_1(x)\cos2x + u_2(x)\sin2x,
      \]
      i.e.
      \[
      \boxed{%
      y_p(x)=\frac{1}{4}\Big(\Big[\tfrac{1}{3}x\cos3x-\tfrac{1}{9}\sin3x+x\cos x-\sin x\Big]\cos2x
      +\Big[\tfrac{1}{3}x\sin3x+\tfrac{1}{9}\cos3x+x\sin x+\cos x\Big]\sin2x\Big).}
      \]
      }}
    \end{center}
\end{enumerate}
