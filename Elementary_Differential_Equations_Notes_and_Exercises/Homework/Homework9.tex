\begin{enumerate}[1.]
  \item Define the \textbf{Laplace transform} of a function $f(t)$, defined for $t \geq 0$.
    \begin{itemize}
      \item[(a)] Using the definition, compute $\laplace\{t * e^{-t}\}$
      \item[(b)] Find the Laplace transform of the piecewise function
                \[f(t) = 
                  \begin{cases}
                  0, & 0 \leq t < 3,\\
                  5, & t \geq 3.
                  \end{cases}
                \]
      \item[(c)] Find $\laplace^{-1}\left\{\frac{s+1}{s^2+4s+5}\right\}$.
    \end{itemize}
    \vspace{1cm}

  \item Solve the IVP
    \[
    y'' + y = \delta(t-\pi), \quad y(0)=0,\ y'(0)=0.
    \]
    \vspace{1cm}
    
  \item Solve the IVP:
      \[
      y'' + y = \sin x, \quad y(0)=0, \ y'(0)=1.
      \]
\end{enumerate}
