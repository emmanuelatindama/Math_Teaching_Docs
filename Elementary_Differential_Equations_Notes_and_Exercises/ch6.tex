\chapter{Systems of First Order Linear \ode{}s}
  \section{Introduction}
    Many problems in science and engineering are naturally modeled as systems of first-order differential equations.
    Instead of one equation in one unknown, we deal with multiple equations involving multiple unknown functions.  

    \begin{weekintro}
    Several connected linear \ode{}s can concisely be written using\index{\ode{}!matrix} matrix\index{matrix} algebra. Studying the matrices involved in the system allows us to better understand the solutions to \ode{}s in qualitative sense (meaning we can sketch their phase portrait) and even write down formulas for their solutions.

    In discussion, we will review the Gauss elimination\index{Gauss elimination}, forming matrix \ode{}s, and computation of eigenvalues\index{eigenvalues} and eigenvectors\index{eigenvectors}.\footnote{You can use \url{https://goo.gl/YQhNkW} to check your steps while studying.}
    \end{weekintro}

    \subsection{Working with \(N \times N\) linear systems}

      \begin{weekintro}
        In principle, the only computational difference in working with matrix \ode{}s of dimensions larger than 3 is that the formulas for 
        computing \(2 \times 2\) and \(3 \times 3\) determinant have to be replaced with the more general \emph{determinant expansion} formula.
        Other than that, we still get \(N\) eigenvalues as roots of \(N\)-degree characteristic polynomial.
        
        Since factoring polynomials by formulas becomes more difficult\footnote{\href{https://goo.gl/iAd3XQ}{Abel's Impossibility Theorem} says
        factoring by formula is impossible for degrees \(> 5\) (quintic).} as we go beyond degree 2.

        For this reason, we'll see that matrices that fall under several special structures (symmetric, skew-symmetric) allow us to infer
        if their eigenvalues are real/complex before computing the eigenvalues.  Other classes of matrices (block-triangular) allow us to
        simplify the characteristic polynomial or even read-off eigenvalues without any additional calculation
        
        While these techniques are essential for \(N \times N\) matrices, they can speed up your work with \(2\times 2\) and \(3 \times 3\) matrices.
      \end{weekintro}

    \begin{definition}
      A \textbf{system of $n$ first-order differential equations} is a collection of equations of the form
      \[
      \frac{dx_1}{dt} = f_1(t, x_1, x_2, \dots, x_n),
      \quad
      \frac{dx_2}{dt} = f_2(t, x_1, x_2, \dots, x_n),
      \quad \dots,
      \quad
      \frac{dx_n}{dt} = f_n(t, x_1, x_2, \dots, x_n),
      \]
      which we may write compactly as
      \[
      \mathbf{x}'(t) = \mathbf{f}(t, \mathbf{x}(t)),
      \]
      where $\mathbf{x}(t) = \begin{bmatrix} x_1(t) \\ x_2(t) \\ \vdots \\ x_n(t) \end{bmatrix}$.
    \end{definition}

    
    \begin{definition}
      A system is called \textbf{linear} if it can be written as
      \[
      \mathbf{x}'(t) = A(t)\mathbf{x}(t) + \mathbf{g}(t),
      \]
      where $A(t)$ is an $n \times n$ matrix (possibly depending on $t$) and $\mathbf{g}(t)$ is a vector function of $t$.
    \end{definition}

    \begin{highlight}
    If $\mathbf{g}(t) \equiv \mathbf{0}$, the system is \textbf{homogeneous}. Otherwise, it is \textbf{nonhomogeneous}.
    \end{highlight}

    
    \begin{theorem}[Existence and Uniqueness]
      Suppose that each $f_i(t, x_1,\dots,x_n)$ is continuous in a region containing $(t_0,\mathbf{x}_0)$ and satisfies a Lipschitz condition in $\mathbf{x}$. Then the IVP
      \[
      \mathbf{x}'(t) = \mathbf{f}(t,\mathbf{x}(t)), \quad \mathbf{x}(t_0)=\mathbf{x}_0,
      \]
      has a unique solution defined in some interval containing $t_0$.
    \end{theorem}

    \textbf{Matrix Form and Solutions}

      For linear systems with constant coefficients:
      \[
      \mathbf{x}'(t) = A\mathbf{x}(t),
      \]
      where $A$ is an $n\times n$ constant matrix, the general solution is expressed in terms of the \textbf{matrix exponential}:
      \[
      \mathbf{x}(t) = e^{At}\mathbf{c}, \quad \mathbf{c} = \mathbf{x}(0).
      \]

      \begin{definition}
        The \textbf{matrix exponential} is defined by the power series
        \[
        e^{At} = I + At + \frac{(At)^2}{2!} + \frac{(At)^3}{3!} + \cdots.
        \]
      \end{definition}

    \textbf{Methods of Solution}
      \begin{enumerate}
        \item \textbf{Eigenvalue-Eigenvector Method:}  
        For constant coefficient systems, find eigenvalues and eigenvectors of $A$ to construct solutions of the form $\mathbf{x}(t) = e^{\lambda t}\mathbf{v}$.
        
        \item \textbf{Fundamental Matrix:}  
        Construct a matrix $\Phi(t)$ whose columns are linearly independent solutions. The general solution is
        \[
        \mathbf{x}(t) = \Phi(t)\mathbf{c}.
        \]
        
        \item \textbf{Variation of Parameters:}  
        For nonhomogeneous systems
        \[
        \mathbf{x}'(t) = A\mathbf{x}(t) + \mathbf{g}(t),
        \]
        one can use
        \[
        \mathbf{x}(t) = \Phi(t)\int \Phi(t)^{-1}\mathbf{g}(t)\, dt.
        \]
      \end{enumerate}

  \section{Homogeneous Linear Systems of Differential Equations}
    \begin{definition}
      A system of $n$ first-order differential equations is called \textbf{homogeneous linear with constant coefficients} if it can be written in matrix form as
      \[
      \mathbf{x}'(t) = A \mathbf{x}(t),
      \]
      where $A$ is an $n \times n$ constant matrix and $\mathbf{x}(t)$ is an $n$-vector of unknown functions.
    \end{definition}
    
    \textbf{General Solution}
    The solution of such systems is based on the \textbf{matrix exponential}:
    \[
    \mathbf{x}(t) = e^{At} \mathbf{c},
    \]
    where $\mathbf{c}$ is a constant vector determined by initial conditions.

    \textbf{Method of Eigenvalues and Eigenvectors}
    \begin{enumerate}
      \item Find the eigenvalues $\lambda_1, \lambda_2, \dots, \lambda_n$ of $A$ by solving
      \[
      \det(A - \lambda I) = 0.
      \]
      \item For each eigenvalue $\lambda$, find the corresponding eigenvector $\mathbf{v}$.
      \item Each pair $(\lambda, \mathbf{v})$ gives a solution of the form
      \[
      \mathbf{x}(t) = e^{\lambda t} \mathbf{v}.
      \]
      \item If the eigenvalues are distinct, the general solution is a linear combination of these solutions.
    \end{enumerate}

    \textbf{Cases}
    \begin{itemize}
      \item \textbf{Distinct Real Eigenvalues:}  
      If $A$ has $n$ distinct real eigenvalues $\lambda_1, \dots, \lambda_n$ with eigenvectors $\mathbf{v}_1, \dots, \mathbf{v}_n$, then
      \[
      \mathbf{x}(t) = c_1 e^{\lambda_1 t} \mathbf{v}_1 + \cdots + c_n e^{\lambda_n t} \mathbf{v}_n.
      \]

      \item \textbf{Complex Eigenvalues:}  
      If $\lambda = \alpha \pm i\beta$ with eigenvector $\mathbf{v} = \mathbf{p} + i \mathbf{q}$, then real solutions can be expressed as
      \[
      e^{\alpha t} \big( c_1 (\mathbf{p} \cos \beta t - \mathbf{q} \sin \beta t) 
      + c_2 (\mathbf{p} \sin \beta t + \mathbf{q} \cos \beta t) \big).
      \]

      \item \textbf{Repeated Eigenvalues:}  
      If an eigenvalue has algebraic multiplicity greater than its geometric multiplicity, generalized eigenvectors are used. In this case, solutions include terms of the form
      \[
      t^k e^{\lambda t}\mathbf{v},
      \]
      where $k$ depends on the size of the Jordan block.
    \end{itemize}

    \textbf{Fundamental Matrix}
    A \textbf{fundamental matrix} $\Phi(t)$ is an $n \times n$ matrix whose columns are linearly independent solutions of the system.  
    \[
    \mathbf{x}(t) = \Phi(t) \mathbf{c}, \quad \Phi(0) = I.
    \]

    \textbf{Phase Plane Analysis (2-D Systems)}


    \begin{weekintro}
      Phase portraits\index{phase portrait!2D} for 2d linear systems can be sketched with the help of eigenvalues and eigenvectors of the system matrices.
      Additionally, general solutions for these systems are found again using eigenvalues and eigenvectors.
    \end{weekintro}

    For a $2 \times 2$ system
    \[
    \mathbf{x}'(t) = A \mathbf{x}(t), \quad A \in \mathbb{R}^{2 \times 2},
    \]
    the qualitative behavior of solutions can be classified by the eigenvalues of $A$:
    \begin{itemize}
      \item Real, distinct, same sign $\implies$ node (stable if negative, unstable if positive).
      \item Real, distinct, opposite signs $\implies$ saddle point.
      \item Real, repeated $\implies$ proper or improper node.
      \item Complex with negative real part $\implies$ stable spiral.
      \item Complex with positive real part $\implies$ unstable spiral.
      \item Purely imaginary $\implies$ center.
    \end{itemize}

    \begin{itemize}
      \item The superposition principle holds: linear combinations of solutions are again solutions.
      \item Homogeneous linear systems always have the trivial solution $\mathbf{x}(t) \equiv \mathbf{0}$.
      \item Initial conditions $\mathbf{x}(t_0) = \mathbf{x}_0$ uniquely determine the constants in the general solution.
    \end{itemize}


  \section{Nonhomogeneous Linear Systems of Differential Equations}

    \begin{definition}
      A system of $n$ first-order differential equations is called \textbf{nonhomogeneous linear with constant coefficients} if it can be written in matrix form as
    \[
    \mathbf{x}'(t) = A \mathbf{x}(t) + \mathbf{f}(t),
    \]
    where $A$ is an $n \times n$ constant matrix, $\mathbf{x}(t)$ is an $n$-vector of unknown functions, and $\mathbf{f}(t)$ is a given non-zero vector function of $t$.
    \end{definition}
    
    \textbf{General Solution}
      The general solution of a nonhomogeneous linear system is the sum of:
      \[
      \mathbf{x}(t) = \mathbf{x}_h(t) + \mathbf{x}_p(t),
      \]
      where
      \begin{itemize}
        \item $\mathbf{x}_h(t)$ is the solution to the associated homogeneous system $\mathbf{x}' = A\mathbf{x}$,
        \item $\mathbf{x}_p(t)$ is a particular solution of the nonhomogeneous system.
      \end{itemize}

    \textbf{Methods for Finding $\mathbf{x}_p(t)$}
    \begin{enumerate}
      \item \textbf{Method of Undetermined Coefficients:}  
      Applicable when $\mathbf{f}(t)$ is a linear combination of functions whose derivatives are of the same type (e.g., polynomials, exponentials, sines, cosines). Assume a trial solution of the same form as $\mathbf{f}(t)$, and solve for unknown coefficients.

      \item \textbf{Variation of Parameters:}  
      General method applicable for any $\mathbf{f}(t)$. If $\Phi(t)$ is a fundamental matrix of the homogeneous system, then a particular solution is
      \[
      \mathbf{x}_p(t) = \Phi(t) \int \Phi^{-1}(t) \mathbf{f}(t)\, dt.
      \]
    \end{enumerate}

    \textbf{Example of General Solution Structure}

    For a 2×2 system:
    \[
    \begin{cases}
    x_1' = 3 x_1 + 4 x_2 + e^{2t},\\
    x_2' = -x_1 + x_2 + \sin t,
    \end{cases}
    \]

    \begin{itemize}
      \item Solve homogeneous system:
      \(\mathbf{x}_h' = A \mathbf{x}_h, \quad A = \begin{bmatrix}3 & 4 \\ -1 & 1\end{bmatrix}\).
      \item Find particular solution $\mathbf{x}_p(t)$ using undetermined coefficients or variation of parameters.
      \item General solution: \(\mathbf{x}(t) = \mathbf{x}_h(t) + \mathbf{x}_p(t)\).
    \end{itemize}


    \begin{itemize}
      \item The superposition principle does \textbf{not} apply to nonhomogeneous solutions alone; it applies only to the homogeneous part.
      \item The particular solution $\mathbf{x}_p(t)$ can be chosen in many ways; any choice that satisfies $\mathbf{x}_p' = A\mathbf{x}_p + \mathbf{f}(t)$ is valid.
      \item Initial conditions $\mathbf{x}(t_0) = \mathbf{x}_0$ determine the constants in the homogeneous solution.
    \end{itemize}


  \section{Matrix Exponential}
    \begin{definition}
      For a square matrix $A \in \mathbb{R}^{n \times n}$, the \textbf{matrix exponential} $e^{At}$ is defined by the power series
      \[
      e^{At} = \sum_{k=0}^{\infty} \frac{(At)^k}{k!} = I + At + \frac{(At)^2}{2!} + \frac{(At)^3}{3!} + \cdots,
      \]
      where $I$ is the $n \times n$ identity matrix.
    \end{definition}
    

    \textbf{Properties}
    \begin{enumerate}
        \item $e^{0} = I$.
        \item $\frac{d}{dt} e^{At} = A e^{At} = e^{At} A$.
        \item If $A$ and $B$ commute ($AB = BA$), then $e^{A+B} = e^A e^B$.
        \item $e^{At} e^{A(-t)} = I$ (inverse property).
        \item If $A$ is diagonalizable, $A = PDP^{-1}$, then
        \[
        e^{At} = P e^{Dt} P^{-1}, \quad e^{Dt} = \text{diag}(e^{\lambda_1 t},\dots,e^{\lambda_n t}).
        \]
    \end{enumerate}

    \textbf{Matrix Exponential and Linear Systems}
    For the homogeneous linear system
    \[
    \mathbf{x}'(t) = A \mathbf{x}(t), \quad \mathbf{x}(0) = \mathbf{x}_0,
    \]
    the solution can be written as
    \[
    \mathbf{x}(t) = e^{At} \mathbf{x}_0.
    \]

    \textbf{Computing $e^{At}$}
    \begin{enumerate}
        \item \textbf{Diagonalizable $A$:} $A = P D P^{-1} \implies e^{At} = P e^{Dt} P^{-1}$.
        \item \textbf{Jordan Form:} If $A$ is not diagonalizable, find its Jordan canonical form $A = P J P^{-1}$, then
        \[
        e^{At} = P e^{Jt} P^{-1}, \quad e^{Jt} \text{ computed block-wise}.
        \]
        \item \textbf{Using Laplace Transform:}  
        If $\mathbf{x}' = A \mathbf{x}$, taking Laplace transforms gives
        \[
        s X(s) - \mathbf{x}_0 = A X(s) \implies X(s) = (sI - A)^{-1} \mathbf{x}_0.
        \]
        Inverting gives $\mathbf{x}(t) = e^{At} \mathbf{x}_0$.
    \end{enumerate}

    
    \begin{example}
      Solve $\mathbf{x}' = \begin{bmatrix} 2 & 1 \\ 0 & 3 \end{bmatrix} \mathbf{x}$, $\mathbf{x}(0) = \begin{bmatrix} 1 \\ 0 \end{bmatrix}$.

      \textbf{Solution:}
      The matrix is upper-triangular, so
      \[
      e^{At} = \begin{bmatrix} e^{2t} & \int_0^t e^{2(t-\tau)} 1 \cdot e^{3\tau} d\tau \\ 0 & e^{3t} \end{bmatrix} 
      = \begin{bmatrix} e^{2t} & e^{3t} - e^{2t} \\ 0 & e^{3t} \end{bmatrix}.
      \]
      Then
      \[
      \mathbf{x}(t) = e^{At} \mathbf{x}_0 = \begin{bmatrix} e^{2t} \\ 0 \end{bmatrix} + \begin{bmatrix} e^{3t} - e^{2t} \\ e^{3t} \end{bmatrix} \begin{bmatrix} 1 \\ 0 \end{bmatrix} = \begin{bmatrix} e^{2t} \\ 0 \end{bmatrix}.
      \]
    \end{example}


    \subsection{Problems Involving Matrices, matrix \ode{}s, eigenvalues, and eigenvectors}
      A second-order differential equation
      \[
      y'' + p(t) y' + q(t) y = g(t)
      \]
      can be rewritten as a system of two first-order equations by introducing a new variable
      \[
      x_1 = y, \quad x_2 = y'.
      \]
      Then,
      \[
      x_1' = x_2, \quad x_2' = -q(t) x_1 - p(t) x_2 + g(t).
      \]
      This system can be written in vector form:
      \[
      \mathbf{x}' = A(t) \mathbf{x} + \mathbf{f}(t), \quad \mathbf{x} = \begin{bmatrix} x_1 \\ x_2 \end{bmatrix}.
      \]

      \begin{example}
        Convert the IVP
        \[
        y'' - 3y' + 2y = 0, \quad y(0)=1, \ y'(0)=0
        \]
        into a first-order system.

        \textbf{Solution:}  

        \begin{align*}
          x_1 &= y, & x_2 &= y', \\
          x_1' &= x_2, & x_2' &= 3x_2 - 2x_1, \\
          \text{System form: } & \mathbf{x}' = \begin{bmatrix} 0 & 1 \\ -2 & 3 \end{bmatrix} \mathbf{x}, \quad \mathbf{x}(0) = \begin{bmatrix} 1 \\ 0 \end{bmatrix}.
        \end{align*}
      \end{example}

      \begin{example}
        Convert the IVP
        \[
        y'' + 4y = \sin t, \quad y(0)=0, \ y'(0)=1
        \]
        into a first-order system.

        \textbf{Solution:}

        \begin{align*}
          x_1 &= y, & x_2 &= y', \\
          x_1' &= x_2, & x_2' &= -4x_1 + \sin t, \\
          \text{System form: } & \mathbf{x}' = \begin{bmatrix} 0 & 1 \\ -4 & 0 \end{bmatrix} \mathbf{x} + \begin{bmatrix} 0 \\ \sin t \end{bmatrix}, \quad \mathbf{x}(0) = \begin{bmatrix} 0 \\ 1 \end{bmatrix}.
        \end{align*}
      \end{example}
