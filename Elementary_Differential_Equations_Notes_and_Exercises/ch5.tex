  \begin{definition}[Laplace Transform]
  The \textbf{Laplace transform} of a function $f(t)$, defined for $t \geq 0$, is given by
  \[
  \laplace\braced*{f(t)}(s) = F(s) = \int_0^\infty e^{-st} f(t)\,dt, \quad s \in \mathbb{C}.
  \]
  Here, $s = \sigma + i \omega$ is a complex variable, and the integral is assumed to converge.
  \end{definition}


  \begin{definition}[Exponential Order]
    A function $f(t)$ is said to be of \emph{exponential order} $\alpha$ if there exist constants $M, \alpha, T > 0$ such that
    \[
    |f(t)| \leq M e^{\alpha t}, \quad \text{for all } t > T.
    \]
    This condition guarantees convergence of the Laplace transform.
  \end{definition}

  \begin{theorem}[Sufficient Condition for Existence]
    If $f(t)$ is piecewise continuous on every finite interval $[0,T]$ and of exponential order $\alpha$, i.e.
    \[
    |f(t)| \leq M e^{\alpha t}, \quad t \geq 0,
    \]
    then the Laplace transform
    \[
    F(s) = \int_0^\infty e^{-st} f(t)\, dt
    \]
    exists for all $s$ with $\Re(s) > \alpha$.
  \end{theorem}

  \begin{theorem}[Boundedness of $F(s)$]
    If $f(t)$ satisfies the above conditions, then $F(s)$ is bounded for $\Re(s) > \alpha$ and is analytic in that region.
  \end{theorem}

  \begin{theorem}[Decay of $F(s)$ as $s \to \infty$]
    If $f(t)$ is piecewise continuous on $[0,\infty)$ and of exponential order, then
    \[
    \lim_{s \to \infty} F(s) = 0.
    \]
  \end{theorem}

  \begin{example}
    For $f(t)=1$, we have $F(s)=1/s \to 0$ as $s \to \infty$.  
    For $f(t)=t$, we have $F(s)=1/s^2 \to 0$ even faster.  
  \end{example}


  \begin{theorem}[Linearity of Laplace Transform]
  If $f(t)$ and $g(t)$ have Laplace transforms $F(s)$ and $G(s)$ respectively, then for constants $a,b \in \mathbb{R}$,
  \[
  \laplace\braced*{af(t) + bg(t)} = aF(s) + bG(s).
  \]
  \end{theorem}


  \begin{theorem}[First Shifting Theorem (frequency shift)]
  If $\laplace\braced*{f(t)} = F(s)$, then
  \[
  \laplace\braced*{e^{at} f(t)} = F(s-a).
  \]
  \end{theorem}

  \begin{example}
    Compute $\laplace\{e^{3t} \sin(2t)\}$.

    \textbf{Solution:}  
    We know $\laplace\{\sin(2t)\} = \frac{2}{s^2 + 4}$. By first shifting theorem:
    \[
    \laplace\{e^{3t}\sin(2t)\} = F(s-3) = \frac{2}{(s-3)^2 + 4}.
    \]
  \end{example}


  \begin{definition}[Unit Step Function]
    The \emph{unit step function} (also called the Heaviside function) is defined as
    \[
    \hstep(t-a) = 
    \begin{cases}
    0, & t < a, \\
    1, & t \geq a,
    \end{cases}
    \]
    for some constant $a \geq 0$.  
    It represents a switch that turns ``on'' at $t=a$.  
    In Laplace transform problems, the unit step function is used to represent piecewise-defined or shifted forcing functions.
  \end{definition}

  \begin{theorem}[Second Shifting Theorem (time shift)]
    If $\laplace\braced*{f(t)} = F(s)$ and $\hstep(t-a)$ is the unit step function, then
    \[
    \laplace\braced*{\hstep(t-a) f(t-a)} = e^{-as} F(s).
    \]
  \end{theorem}

  \begin{example}
    Compute $\laplace\{\hstep(t-2) \cdot (t-2)\}$.

    \textbf{Solution:}  
    We know $\laplace\{t\} = \frac{1}{s^2}$. By second shifting theorem:
    \[
    \laplace\{\hstep(t-2)(t-2)\} = e^{-2s} \cdot \frac{1}{s^2}.
    \]
  \end{example}

  
  \begin{example}
    Find the Laplace transform of the piecewise function
    \[
    f(t) = 
    \begin{cases}
    0, & 0 \leq t < 3,\\
    5, & t \geq 3.
    \end{cases}
    \]
    \textbf{Solution:}
    We rewrite using the unit step function:
    \[
    f(t) = 5\,\hstep(t-3).
    \]
    Thus,
    \begin{align*}
      \laplace\{f(t)\} &= 5 \laplace\{\hstep(t-3)\}\\
      &= \frac{5 e^{-3s}}{s}.
    \end{align*}
    \[
    \boxed{F(s) = \tfrac{5 e^{-3s}}{s}}
    \]
  \end{example}

  \begin{example}
    Find the Laplace transform of
    \[
    f(t) = 
    \begin{cases}
    t, & 0 \leq t < 1,\\
    1, & t \geq 1.
    \end{cases}
    \]
    \textbf{Solution:}
    We rewrite in terms of $\hstep(t)$:
    \[
    f(t) = t(1-\hstep(t-1)) + 1\cdot \hstep(t-1).
    \]
    Simplify:
    \[
    f(t) = t - t \hstep(t-1) + \hstep(t-1).
    \]
    Take Laplace transform:
    \begin{align*}
    F(s) &= \laplace\{t\} - \laplace\{\hstep(t-1)(t)\} + \laplace\{\hstep(t-1)\}\\
    &= \frac{1}{s^2} - e^{-s}\laplace\{t+1\} + \frac{e^{-s}}{s}.
    \end{align*}
    Since $\laplace\{t+1\}=\frac{1}{s^2}+\frac{1}{s}$, we get
    \[
    F(s)=\frac{1}{s^2} - e^{-s}\Big(\frac{1}{s^2}+\frac{1}{s}\Big) + \frac{e^{-s}}{s}.
    \]
    Simplify:
    \[
    \boxed{F(s)=\frac{1}{s^2} - \frac{e^{-s}}{s^2}}
    \]
  \end{example}

  \begin{theorem}[Differentiation in Time Domain]
    \[
    \laplace\braced*{f'(t)} = sF(s) - f(0),
    \]
    \[
    \laplace\braced*{f''(t)} = s^2 F(s) - sf(0) - f'(0).
    \]
  \end{theorem}

  \begin{example}
    Find $\laplace\{y''(t)\}$ if $y(t)$ satisfies $y(0)=2$, $y'(0)=1$ and $\laplace\{y(t)\}=Y(s)$.

    \textbf{Solution:}  
    By differentiation in time domain:
    \[
    \laplace\{y''(t)\} = s^2 Y(s) - s y(0) - y'(0) = s^2 Y(s) - 2s - 1.
    \]
  \end{example}


  \begin{theorem}[Differentiation in $s$-Domain]
    \[
    \laplace\braced*{t f(t)} = -F'(s).
    \]
  \end{theorem}

  \begin{example}
    Compute $\laplace\{t \cos(2t)\}$.

    \textbf{Solution:}  
    We know $\laplace\{\cos(2t)\} = \frac{s}{s^2 + 4}$. By differentiation in $s$-domain:
    \[
    \laplace\{t \cos(2t)\} = -\frac{d}{ds} \frac{s}{s^2 + 4} = \frac{s^2 - 4}{(s^2 + 4)^2}.
    \]
  \end{example}


  \begin{theorem}[Integration in Time Domain]
    \[
    \laplace\left\{\int_0^t f(\tau)\,d\tau\right\} = \frac{F(s)}{s}.
    \]
  \end{theorem}

  \begin{example}
    Compute $\laplace\left\{\int_0^t e^{2\tau} d\tau\right\}$.

    \textbf{Solution:}  
    \[
    F(s) = \laplace\{e^{2t}\} = \frac{1}{s-2} \implies 
    \laplace\left\{\int_0^t e^{2\tau} d\tau\right\} = \frac{1}{s(s-2)}.
    \]
  \end{example}


  \begin{theorem}[Multiplication by Exponential / Convolution Property]
    \[
    \laplace\{f(t) * g(t)\} = F(s) G(s),
    \]
    where $f*g = \int_0^t f(\tau) g(t-\tau)\,d\tau$ is the convolution of $f$ and $g$.
  \end{theorem}

  \begin{example}
    Compute $\laplace\{t * e^{-t}\}$.

    \textbf{Solution:}  
    \[
    \laplace\{t\} = \frac{1}{s^2},\quad \laplace\{e^{-t}\} = \frac{1}{s+1}.
    \]
    By convolution theorem:
    \[
    \laplace\{t * e^{-t}\} = \frac{1}{s^2} \cdot \frac{1}{s+1} = \frac{1}{s^2 (s+1)}.
    \]
  \end{example}

  \begin{definition}
    The \textbf{Dirac delta function}, denoted by $\delta(t-a)$, is not a function in the traditional sense but a \emph{generalized function} or \emph{distribution}. It is defined by the following properties:
    \begin{enumerate}
      \item $\delta(t-a) = 0$ for all $t \neq a$,
      \item $\displaystyle \int_{-\infty}^\infty \delta(t-a)\, f(t)\, dt = f(a)$ for any continuous test function $f(t)$.
    \end{enumerate}
  \end{definition}

  \begin{highlight}
    Intuitively, $\delta(t-a)$ can be thought of as an ``infinite spike'' at $t=a$ with unit area under the curve.  
  \end{highlight}

  \begin{theorem}[Laplace Transform of Delta]
    For $a \geq 0$, the Laplace transform is
    \[
    \laplace\{\delta(t-a)\} = e^{-as}.
    \]
  \end{theorem}

  
  \begin{example}
    Solve the IVP
    \[
    y'' + y = \delta(t-\pi), \quad y(0)=0,\ y'(0)=0.
    \]

    \textbf{Solution:}

    Step 1: Laplace transform.
    \[
    (s^2Y(s)) + Y(s) = e^{-\pi s}.
    \]

    Step 2: Solve for $Y(s)$.
    \[
    Y(s) = \frac{e^{-\pi s}}{s^2+1}.
    \]

    Step 3: Inverse Laplace.
    \[
    \laplace^{-1}\!\left\{\frac{1}{s^2+1}\right\} = \sin t.
    \]

    Step 4: Apply shifting theorem.
    \[
    y(t) = u(t-\pi)\sin(t-\pi).
    \]

    \textbf{Final Answer:}
    \[
    \boxed{y(t) = u(t-\pi)\sin(t-\pi)}.
    \]
  \end{example}


  \begin{example}
    Solve the IVP
    \[
    y' + y = \delta(t-2), \quad y(0)=0.
    \]

    \textbf{Solution:}

    Step 1: Laplace transform.
    \[
    (sY(s)) + Y(s) = e^{-2s}.
    \]
    \[
    (s+1)Y(s) = e^{-2s}.
    \]

    Step 2: Solve for $Y(s)$.
    \[
    Y(s) = \frac{e^{-2s}}{s+1}.
    \]

    Step 3: Inverse Laplace using shift theorem.
    \[
    \laplace^{-1}\!\left\{\frac{1}{s+1}\right\} = e^{-t}.
    \]
    \[
    y(t) = u(t-2)e^{-(t-2)}.
    \]

    \textbf{Final Answer:}
    \[
    \boxed{y(t) = u(t-2)e^{-(t-2)}}.
    \]
  \end{example}

  \begin{theorem}[Initial Value Theorem]
    Let $F(s) = \laplace\braced*{f(t)}$. Then
    \[
    \lim_{t \to 0^+} f(t) = \lim_{s \to \infty} sF(s),
    \]
    provided the limit exists.
  \end{theorem}


  \begin{theorem}[Final Value Theorem]
    Let $F(s) = \laplace\braced*{f(t)}$. Then
    \[
    \lim_{t \to \infty} f(t) = \lim_{s \to 0^+} sF(s),
    \]
    provided the poles of $sF(s)$ lie strictly in $\Re(s)<0$.
  \end{theorem}


  \begin{example}[Transform of a constant]
    \[
    f(t) = 1, \quad F(s) = \int_0^\infty e^{-st} \,dt.
    \]
    Compute:
    \[
    F(s) = \left[ \frac{-1}{s} e^{-st} \right]_0^\infty = \frac{1}{s}, \quad \Re(s) > 0.
    \]
  \end{example}

  \begin{example}[Transform of $t$]
    \[
    f(t) = t, \quad F(s) = \int_0^\infty t e^{-st} \,dt.
    \]
    Using integration by parts:
    \[
    u = t, \, dv = e^{-st}dt \quad \Rightarrow \quad du = dt, \, v = -\frac{1}{s} e^{-st}.
    \]
    Then
    \[
    F(s) = \left[-\frac{t}{s} e^{-st}\right]_0^\infty + \int_0^\infty \frac{1}{s} e^{-st}\,dt.
    \]
    First term vanishes, so
    \[
    F(s) = \frac{1}{s^2}, \quad \Re(s) > 0.
    \]
  \end{example}

  \begin{example}[Transform of $e^{at}$]
    \[
    f(t) = e^{at}, \quad F(s) = \int_0^\infty e^{-st} e^{at}\,dt = \int_0^\infty e^{-(s-a)t}\,dt.
    \]
    Provided $\Re(s) > a$, we get
    \[
    F(s) = \frac{1}{s-a}.
    \]
  \end{example}

  \section{Inverse Laplace Transforms}

    \begin{definition}[Inverse Laplace Transform]
      Given a Laplace transform $F(s)$ of a function $f(t)$, the \textbf{inverse Laplace transform} is the operation that recovers $f(t)$ from $F(s)$.
      It is denoted by
      \[
      f(t) = \laplace^{-1}\braced*{F(s)}.
      \]
    \end{definition}


    \begin{theorem}[Uniqueness of Inverse Laplace Transform]
    If $F(s)$ is the Laplace transform of a piecewise continuous function $f(t)$ of exponential order, then the inverse Laplace transform exists and is unique almost everywhere.
    \end{theorem}

    \textbf{Properties of the Inverse Transform}

    \begin{itemize}
      \item \textbf{Linearity:} $\laplace^{-1}\braced*{a F(s) + b G(s)} = a \laplace^{-1}\braced*{F(s)} + b \laplace^{-1}\braced*{G(s)}$.
      \item \textbf{First Shifting Theorem:} $\laplace^{-1}\braced*{F(s-a)} = e^{at} f(t)$.
      \item \textbf{Second Shifting Theorem:} $\laplace^{-1}\braced*{e^{-as} F(s)} = \hstep(t-a) f(t-a)$, where $\hstep(t-a)$ is the unit step function.
      \item \textbf{Differentiation in $s$:} $\laplace^{-1}\braced*{-\frac{d}{ds} F(s)} = t f(t)$.
    \end{itemize}

    \textbf{Methods of Finding Inverse Laplace Transforms}

    \begin{enumerate}
      \item Using a \textbf{reference table} of common Laplace transforms.
      \item \textbf{Partial fraction decomposition} for rational functions of $s$.
      \item Using the \textbf{first and second shifting theorems}.
      \item \textbf{Convolution theorem:} $\laplace^{-1}\braced*{F(s)G(s)} = (f*g)(t) = \int_0^t f(\tau) g(t-\tau) \, d\tau$.
    \end{enumerate}

    \begin{example}
      Find $\laplace^{-1}\left\{\frac{1}{s-3}\right\}$.

      \textbf{Solution:}
      
      From the table, $\laplace^{-1}\left\{\frac{1}{s-a}\right\} = e^{at}$, hence
      \[
      \laplace^{-1}\left\{\frac{1}{s-3}\right\} = e^{3t}.
      \]
    \end{example}


    \begin{example}
      Find $\laplace^{-1}\left\{\frac{2}{s^2+9}\right\}$.

      \textbf{Solution:}
      
      From the table, $\laplace^{-1}\left\{\frac{b}{s^2+b^2}\right\} = \sin(bt)$, so
      \[
      \laplace^{-1}\left\{\frac{2}{s^2+9}\right\} = \frac{2}{3} \sin(3t).
      \]
    \end{example}


    \begin{example}
      Find $\laplace^{-1}\left\{\frac{s+1}{s^2+4s+5}\right\}$.

      \textbf{Solution:}
      
      Complete the square: $s^2+4s+5 = (s+2)^2 + 1$.
      
      Rewrite numerator: $s+1 = (s+2) - 1$.

      Split into two terms:
      \[
      \frac{s+1}{(s+2)^2 + 1} = \frac{s+2}{(s+2)^2 +1} - \frac{1}{(s+2)^2+1}.
      \]  
      From the table:
      \[
      \laplace^{-1}\left\{\frac{s+2}{(s+2)^2 +1}\right\} = e^{-2t}\cos t, \quad
      \laplace^{-1}\left\{\frac{1}{(s+2)^2 +1}\right\} = e^{-2t}\sin t.
      \]  
      Hence,
      \[
      \laplace^{-1}\left\{\frac{s+1}{s^2+4s+5}\right\} = e^{-2t} (\cos t - \sin t).
      \]
    \end{example}

    \begin{example}
      Solve the IVP
      \[
      y' - 3y = 6,\qquad y(0)=2,
      \]
      using Laplace transforms.

      \textbf{Solution.} Let $Y(s)=\mathcal{L}\{y(t)\}(s)$.
      
      Taking Laplace transforms and using $\mathcal{L}\{y'\}=sY(s)-y(0)$ gives
      \begin{align*}
      \mathcal{L}\{y'\}-3\mathcal{L}\{y\} &= \mathcal{L}\{6\} \\
      \big(sY(s)-y(0)\big) - 3Y(s) &= \frac{6}{s} \\
      (s-3)Y(s) - 2 &= \frac{6}{s} \\
      (s-3)Y(s) &= \frac{6}{s} + 2 = \frac{6+2s}{s} \\
      Y(s) &= \frac{6+2s}{s(s-3)} = \frac{2s+6}{s(s-3)}.
      \end{align*}

      Perform partial fraction decomposition: seek $A,B$ with
      \[
      \frac{2s+6}{s(s-3)}=\frac{A}{s}+\frac{B}{s-3}.
      \]
      Equating numerators,
      \begin{align*}
      2s+6 &= A(s-3)+Bs = (A+B)s - 3A.
      \end{align*}
      Compare coefficients:
      \begin{align*}
      A+B &= 2,\\
      -3A &= 6 \;\Rightarrow\; A = -2,\qquad B = 2 - A = 4.
      \end{align*}
      Thus
      \[
      Y(s) = -\frac{2}{s} + \frac{4}{s-3}.
      \]

      Taking inverse Laplace termwise yields
      \begin{align*}
      y(t) &= -2\mathcal{L}^{-1}\!\left\{\frac{1}{s}\right\} + 4\mathcal{L}^{-1}\!\left\{\frac{1}{s-3}\right\} \\
          &= -2 + 4e^{3t}.
      \end{align*}

      \[
      \boxed{\,y(t) = -2 + 4e^{3t}\,}
      \]
    \end{example}


    \begin{example}
      Solve the IVP:
      \[
      y'' - 4y' + 4y = 0,\qquad y(0)=1,\; y'(0)=0,
      \]
      using Laplace transforms.

      \textbf{Solution.}
      
      Let $Y(s)=\mathcal{L}\{y(t)\}(s)$.

      Taking Laplace transforms and using
      $$\mathcal{L}\{y''\}=s^2Y(s)-sy(0)-y'(0) \:\text{ and }\: \mathcal{L}\{y'\}=sY(s)-y(0) \:\text{ gives }$$

      \begin{align*}
      \mathcal{L}\{y''\} - 4\mathcal{L}\{y'\} + 4\mathcal{L}\{y\} &= 0\\
      \big(s^2Y(s)-s\cdot 1 - 0\big) - 4\big(sY(s)-1\big) + 4Y(s) &= 0\\
      \big(s^2 - 4s + 4\big)Y(s) - s + 4 &= 0\\
      (s-2)^2 Y(s) &= s - 4\\
      Y(s) &= \frac{s-4}{(s-2)^2}.
      \end{align*}

      Rewrite the numerator to prepare for inversion:
      \begin{align*}
      \frac{s-4}{(s-2)^2} &= \frac{(s-2)-2}{(s-2)^2}
      = \frac{s-2}{(s-2)^2} - \frac{2}{(s-2)^2} \\
      &= \frac{1}{s-2} - \frac{2}{(s-2)^2}.
      \end{align*}

      Take inverse Laplace termwise:
      \begin{align*}
      y(t) &= \mathcal{L}^{-1}\!\left\{\frac{1}{s-2}\right\}
            - 2\mathcal{L}^{-1}\!\left\{\frac{1}{(s-2)^2}\right\}\\
          &= e^{2t} - 2t e^{2t}\\
          &= (1-2t)e^{2t}.
      \end{align*}

      \[

  \subsection*{Solving IVPs with Unit Step Forcing Functions}

    \begin{example}
      Solve the IVP:
      \[
      y'' + y = \hstep(t-\pi), \quad y(0)=0,\ y'(0)=0.
      \]

      \textbf{Solution:}

      Step 1: Laplace transform of both sides.
      \begin{align*}
      \laplace\{y''\} + \laplace\{y\} &= \laplace\{\hstep(t-\pi)\},\\
      (s^2 Y(s) - s y(0) - y'(0)) + Y(s) &= \frac{e^{-\pi s}}{s}.
      \end{align*}
      Substitute $y(0)=0$, $y'(0)=0$:
      \[
      (s^2+1)Y(s) = \frac{e^{-\pi s}}{s}.
      \]

      Step 2: Solve for $Y(s)$.
      \[
      Y(s) = \frac{e^{-\pi s}}{s(s^2+1)}.
      \]

      Step 3: Inverse Laplace.
      \[
      \laplace^{-1}\!\left\{\frac{1}{s(s^2+1)}\right\} = 1 - \cos t.
      \]
      Hence,
      \[
      y(t) = \hstep(t-\pi)\big(1 - \cos(t-\pi)\big).
      \]

      \textbf{Final Answer:}
      \[
      \boxed{y(t) = \hstep(t-\pi)\big(1 - \cos(t-\pi)\big)}
      \]
    \end{example}

    \begin{example}
      Solve the IVP:
      \[
      y'' + y' = \hstep(t-2), \quad y(0)=0,\ y'(0)=1.
      \]

      \textbf{Solution:}

      Step 1: Laplace transform.
      \begin{align*}
      (s^2 Y(s) - sy(0) - y'(0)) + (sY(s) - y(0)) &= \frac{e^{-2s}}{s},\\
      (s^2 + s)Y(s) - 1 &= \frac{e^{-2s}}{s}.
      \end{align*}

      Step 2: Solve for $Y(s)$.
      \[
      Y(s) = \frac{1}{s(s+1)} + \frac{e^{-2s}}{s^2(s+1)}.
      \]

      Step 3: Inverse Laplace of first term.
      \[
      \frac{1}{s(s+1)} = \frac{1}{s} - \frac{1}{s+1}
      \quad \implies \quad \laplace^{-1} = 1 - e^{-t}.
      \]

      Step 4: Inverse Laplace of second term.
      \[
      \frac{1}{s^2(s+1)} = \frac{1}{s^2} - \frac{1}{s} + \frac{1}{s+1}.
      \]
      Inverse Laplace:
      \[
      \laplace^{-1} = t - 1 + e^{-t}.
      \]
      Apply shifting theorem:
      \[
      \laplace^{-1}\Big\{\frac{e^{-2s}}{s^2(s+1)}\Big\} = \hstep(t-2)\big((t-2) - 1 + e^{-(t-2)}\big).
      \]

      Step 5: Combine.
      \[
      y(t) = (1 - e^{-t}) + \hstep(t-2)\big((t-3) + e^{-(t-2)}\big).
      \]

      \textbf{Final Answer:}
      \[
      \boxed{y(t) = 1 - e^{-t} + \hstep(t-2)\big((t-3) + e^{-(t-2)}\big)}
      \]
    \end{example}

    \subsection*{Time-shifts, steps, and impulse responses}

    \begin{weekintro}
      Laplace transform is extremely useful in resolving problems that\index{Laplace transform!time shift} feature \textbf{piecewise}-defined functions.

      The main tool for this is the time-shift rule. More advanced courses will further develop your knowledge of this part, here we will only
      familiarize ourselves with basic mechanics of time-shifts.

      Dirac impulse\index{impulse (Dirac)} \(\delta(t)\) and Heaviside step \index{step (Heaviside)} \(\hstep(t)\) are two 
      functions\footnote{Dirac \(\delta\) is technically \emph{not} a function, but rather a \emph{distribution} or a \emph{generalized} function. 

      However, using Laplace transform we can work with it as if it was a function.} that you perhaps haven't seen before.
      They are featured prominently in engineering analysis of dynamical systems, as they allow an engineer to investigate the dynamics of the system without detailed modeling of the physical components inside (so-called ``black-box modeling'').
      
      Solutions of \ode{}s with step/impulse inputs and zero initial conditions are called \textbf{step/impulse responses}.
    \end{weekintro}
